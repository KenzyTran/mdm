       ```python
       if gate == "SELL":
           # Rank open positions by RS — close weakest, keep strongest.
           # Positions already scheduled for mdm_sell exit are excluded.
           existing = {
               id(sx.position)
               for sx in self._scheduled_exits
               if sx.reason == "mdm_sell"
           }
           eligible = [
               pos for pos in self.book.slots.open_positions
               if id(pos) not in existing and bar_idx >= pos.earliest_sell_bar
           ]
           if eligible:
               import math as _math
               n_keep = _math.ceil(len(eligible) * self.cfg.sell_retain_pct)
               # Sort by RS descending; NaN/missing treated as -inf (weakest)
               def _rs_sort_key(pos):
                   val = self._rs_lookup.get((pos.ticker, date))
                   if val is None or (isinstance(val, float) and _math.isnan(val)):
                       return float("-inf")
                   return val
               ranked = sorted(eligible, key=_rs_sort_key, reverse=True)
               # Keep top n_keep, schedule exit for the rest
               to_close = ranked[n_keep:]
               for pos in to_close:
                   self._scheduled_exits.append(
                       _ScheduledExit(
                           position=pos,
                           fill_bar_idx=bar_idx + 1,
                           reason="mdm_sell",
                       )
                   )
       ```

       Note: `import math` is already imported at top of file (used in `_update_rs_streaks`). Use `math.ceil` and `math.isnan` directly — do NOT add a local `_math` alias. The snippet above is illustrative; use the existing `math` import.
