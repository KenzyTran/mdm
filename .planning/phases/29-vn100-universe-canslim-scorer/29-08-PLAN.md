    ```python
    """CanslimScorer — end-to-end per-(date,ticker) scoring (CANS-11, D-09..D-11)."""
    from dataclasses import dataclass
    from datetime import date
    from typing import Optional
    import numpy as np
    import pandas as pd

    from strategies.canslim.config import CanslimConfig
    from strategies.canslim.universe import UniverseLoader
    from strategies.canslim.sectors import SectorRouter
    from strategies.canslim.rules.fundamental import compute_fundamentals
    from strategies.canslim.rules.technical import compute_n, compute_s
    from strategies.canslim.rules.rs import compute_rs_ratings
    from strategies.canslim.rules.flow import compute_i
    from strategies.canslim.rules.liquidity import compute_liq

    NUM_BOOLEAN_RULES = 9  # c, c+, a, a+, n, s, l, i, liq

    @dataclass
    class CanslimScorer:
        config: CanslimConfig
        universe_loader: UniverseLoader
        sector_router: SectorRouter
        pg_engine: object
        mysql_engine: object

        def score(self, as_of_date: date) -> pd.DataFrame:
            tickers = sorted(self.universe_loader.get(as_of_date))
            tickers = [t for t in tickers if not self.sector_router.is_excluded(t)]
            if not tickers:
                return self._empty_frame()

            # Load the full price panel for the universe (one query).
            panel = self._load_panel(tickers, as_of_date)

            # RS ratings once, universe-wide.
            rs = compute_rs_ratings(panel, as_of_date, tickers, self.config)

            rows = []
            for t in tickers:
                ohlcv = panel[panel["stockcode"].str.upper() == t].sort_values("tradingdate")
                c, cp, a, ap = compute_fundamentals(
                    t, as_of_date, self.config, self.sector_router, self.mysql_engine,
                )
                n = compute_n(ohlcv, as_of_date, self.config)
                s = compute_s(ohlcv, as_of_date, self.config)
                i = compute_i(t, as_of_date, self.config, self.pg_engine)
                liq = compute_liq(ohlcv, as_of_date, self.config)
                rs_val = float(rs.get(t, np.nan))
                l = (not np.isnan(rs_val)) and (rs_val >= self.config.l_rs_threshold)
                passes = [c, cp, a, ap, n, s, l, i, liq]
                # LOCKED composite formula: 70% * boolean pass ratio + 30% * RS rating (0..100)
                bool_component = 100.0 * sum(passes) / NUM_BOOLEAN_RULES
                rs_component = 0.0 if np.isnan(rs_val) else rs_val
                composite = 0.70 * bool_component + 0.30 * rs_component
                rows.append({
                    "date": as_of_date, "ticker": t,
                    "sector": self.sector_router.route(t),
                    "c_pass": c, "c_plus_pass": cp, "a_pass": a, "a_plus_pass": ap,
                    "n_pass": n, "s_pass": s, "l_pass": l, "i_pass": i, "liq_pass": liq,
                    "rs_rating": rs_val, "score": composite,
                })
            return pd.DataFrame(rows)

        def _load_panel(self, tickers, as_of_date) -> pd.DataFrame:
            sql = """
                SELECT stockcode, tradingdate, closeindex, highestindex, totalvol
                FROM stock_eod
                WHERE stockcode = ANY(%(t)s) AND tradingdate <= %(d)s
                ORDER BY stockcode, tradingdate
            """
            return pd.read_sql(sql, self.pg_engine, params={"t": tickers, "d": as_of_date})

        def _empty_frame(self) -> pd.DataFrame:
            cols = ["date","ticker","sector","c_pass","c_plus_pass","a_pass","a_plus_pass",
                    "n_pass","s_pass","l_pass","i_pass","liq_pass","rs_rating","score"]
            return pd.DataFrame({c: pd.Series(dtype="object") for c in cols})
    ```
