    ```python
    """Fundamental rules C, C+, A, A+ (CANS-01..04, D-07, D-11)."""
    from datetime import date
    from typing import Tuple, List, Optional
    import json
    from pathlib import Path
    import pandas as pd

    from strategies.canslim.config import CanslimConfig
    from strategies.canslim.sectors import SectorRouter

    _SCHEMA_LOCK = None

    def _schema() -> dict:
        global _SCHEMA_LOCK
        if _SCHEMA_LOCK is None:
            p = Path(".planning/phases/29-vn100-universe-canslim-scorer/schema_lock.json")
            _SCHEMA_LOCK = json.loads(p.read_text())
        return _SCHEMA_LOCK["locked"]

    def _load_quarters(mysql_engine, table: str, value_col: str, ticker: str,
                       as_of_date: date, publish_col: str) -> pd.DataFrame:
        """Load published quarterly rows up to as_of_date. Enforces D-11 look-ahead guard."""
        sql = f"""
            SELECT period_end, {value_col} AS value, {publish_col} AS publish_date
            FROM {table}
            WHERE stockcode = %(t)s AND {publish_col} <= %(d)s
            ORDER BY period_end DESC
            LIMIT 16
        """
        return pd.read_sql(sql, mysql_engine, params={"t": ticker, "d": as_of_date})

    def compute_c(values: List[float], threshold: float) -> bool:
        """C: current quarter YoY vs 4 quarters ago >= threshold."""
        if len(values) < 5:
            return False
        yoy = (values[0] - values[4]) / abs(values[4]) if values[4] != 0 else 0
        return yoy >= threshold

    def compute_c_plus(values: List[float]) -> bool:
        """C+: current quarter YoY growth exceeds mean of prior two quarters' YoY."""
        if len(values) < 7:
            return False
        def yoy(i): return (values[i] - values[i+4]) / abs(values[i+4]) if values[i+4] else 0
        return yoy(0) > (yoy(1) + yoy(2)) / 2

    def compute_a(values: List[float], threshold: float) -> bool:
        """A: 3yr TTM-sum CAGR >= threshold. Needs >=12 quarters (current TTM + 2yr ago TTM)."""
        if len(values) < 12:
            return False
        ttm_now = sum(values[0:4])
        ttm_2y_ago = sum(values[8:12])
        if ttm_2y_ago <= 0 or ttm_now <= 0:
            return False
        cagr = (ttm_now / ttm_2y_ago) ** (1/2) - 1
        return cagr >= threshold

    def compute_a_plus(annual_values: List[float]) -> bool:
        """A+: last 3 annual values all > 0. Annual computed as rolling TTM from quarterly."""
        if len(annual_values) < 3:
            return False
        return all(v > 0 for v in annual_values[:3])

    def compute_fundamentals(
        ticker: str, as_of_date: date, config: CanslimConfig,
        sector_router: SectorRouter, mysql_engine,
    ) -> Tuple[bool, bool, bool, bool]:
        sector = sector_router.route(ticker)
        if sector in ("ctck", "insurance"):
            return (False, False, False, False)

        locked = _schema()
        publish_col = locked["publish_date_column"]
        if sector == "bank":
            table = "is_quarter_bank"
            value_col = locked["is_quarter_bank_ppop_column"]
        else:
            table = "is_quarter_nonbank"
            value_col = locked["is_quarter_nonbank_eps_column"]

        df = _load_quarters(mysql_engine, table, value_col, ticker, as_of_date, publish_col)
        if df.empty:
            return (False, False, False, False)
        values = df["value"].astype(float).tolist()

        # Derive annual-ish TTM rollup from quarterly (A+ uses rolling TTMs offset by 4q)
        annuals = []
        for offset in (0, 4, 8):
            if len(values) >= offset + 4:
                annuals.append(sum(values[offset:offset+4]))

        return (
            compute_c(values, config.c_threshold),
            compute_c_plus(values),
            compute_a(values, config.a_threshold),
            compute_a_plus(annuals),
        )
    ```
