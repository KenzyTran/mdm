    """I rule — foreign net-buy flow (CANS-08). Pre-2022 fallback per 29-RESEARCH."""
    from datetime import date
    import pandas as pd

    from strategies.canslim.config import CanslimConfig

    FOREIGN_DATA_START = date(2022, 4, 7)

    def compute_i(ticker: str, as_of_date: date, config: CanslimConfig, pg_engine) -> bool:
        if as_of_date < FOREIGN_DATA_START:
            # Fallback: stock_foreign_eod has no data before 2022-04-07.
            # Research + D-?? documented decision: pass the I rule by default.
            return True
        sql = """
            SELECT net_buy FROM (
                SELECT net_buy_value AS net_buy
                FROM stock_foreign_eod
                WHERE stockcode = %(t)s AND tradingdate < %(d)s
                ORDER BY tradingdate DESC
                LIMIT %(n)s
            ) q
        """
        df = pd.read_sql(sql, pg_engine, params={
            "t": ticker.upper(), "d": as_of_date, "n": config.i_lookback_days,
        })
        if df.empty:
            return False
        return float(df["net_buy"].sum()) > 0
    ```

    """Liquidity rule (CANS-09). 20d median turnover >= config threshold."""
    import pandas as pd
    from strategies.canslim.config import CanslimConfig

    def compute_liq(ohlcv: pd.DataFrame, as_of_date, config: CanslimConfig) -> bool:
        df = ohlcv[ohlcv["tradingdate"] <= as_of_date].tail(20)
        if len(df) < 20:
            return False
        turnover = (df["closeindex"] * df["totalvol"]).median()
        return float(turnover) >= config.liquidity_min_turnover_vnd
    ```
