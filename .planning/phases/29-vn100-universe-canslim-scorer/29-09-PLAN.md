    ```python
    """Baseline loader for rank_top_stocks.diem_canslim (CANS-12, D-14..D-16)."""
    from datetime import date
    from typing import List, Tuple
    import pandas as pd

    def load_diem_canslim_top10(mysql_engine, as_of_date: date) -> List[Tuple[str, float]]:
        """Return top-10 tickers by diem_canslim for exact date; empty list if no data."""
        sql = """
            SELECT stockcode, diem_canslim
            FROM rank_top_stocks
            WHERE tradingdate = %(d)s AND diem_canslim IS NOT NULL
            ORDER BY diem_canslim DESC
            LIMIT 10
        """
        df = pd.read_sql(sql, mysql_engine, params={"d": as_of_date})
        return list(zip(df["stockcode"].str.upper(), df["diem_canslim"].astype(float)))

    def available_baseline_dates(mysql_engine, start: date, end: date) -> List[date]:
        sql = """
            SELECT DISTINCT tradingdate
            FROM rank_top_stocks
            WHERE tradingdate BETWEEN %(s)s AND %(e)s AND diem_canslim IS NOT NULL
            ORDER BY tradingdate
        """
        df = pd.read_sql(sql, mysql_engine, params={"s": start, "e": end})
        return df["tradingdate"].tolist()
    ```
