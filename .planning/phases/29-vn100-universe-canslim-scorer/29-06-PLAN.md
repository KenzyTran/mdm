    """Technical rules N (CANS-05) and S (CANS-06 helper)."""
    import pandas as pd
    from strategies.canslim.config import CanslimConfig

    def compute_n(ohlcv: pd.DataFrame, as_of_date, config: CanslimConfig) -> bool:
        """N: close within config.n_within_high of the 252-day high.
        ohlcv must have columns ['tradingdate','closeindex','highestindex'] sorted ascending.
        """
        df = ohlcv[ohlcv["tradingdate"] <= as_of_date].tail(config.min_history_days)
        if len(df) < config.min_history_days:
            return False
        high_252 = df["highestindex"].max()
        close = df.iloc[-1]["closeindex"]
        return close >= (1 - config.n_within_high) * high_252

    def compute_s(ohlcv: pd.DataFrame, as_of_date, config: CanslimConfig) -> bool:
        """S: today's volume >= s_vol_mult * avgvol50."""
        df = ohlcv[ohlcv["tradingdate"] <= as_of_date].tail(51)
        if len(df) < 51:
            return False
        avg50 = df.iloc[:-1]["totalvol"].mean()
        today_vol = df.iloc[-1]["totalvol"]
        return today_vol >= config.s_vol_mult * avg50
    ```

    """RS rating L rule (CANS-07). Formula: 0.4*ROC63 + 0.2*ROC126 + 0.2*ROC189 + 0.2*ROC252.
    Percentile-ranked within the active universe at as_of_date.
    """
    from datetime import date
    from typing import Iterable
    import numpy as np
    import pandas as pd

    from strategies.canslim.config import CanslimConfig

    def _roc(prices: pd.Series, lookback: int) -> float:
        if len(prices) <= lookback:
            return np.nan
        past = prices.iloc[-lookback - 1]
        now = prices.iloc[-1]
        if past <= 0:
            return np.nan
        return (now / past) - 1.0

    def _raw_rs(prices: pd.Series, config: CanslimConfig) -> float:
        if len(prices) < config.min_history_days:
            return np.nan
        total = 0.0
        for w, lb in zip(config.rs_weights, config.rs_roc_days):
            r = _roc(prices, lb)
            if np.isnan(r):
                return np.nan
            total += w * r
        return total

    def compute_rs_ratings(
        panel: pd.DataFrame, as_of_date: date,
        universe: Iterable[str], config: CanslimConfig,
    ) -> pd.Series:
        """panel: long-format DataFrame with columns ['stockcode','tradingdate','closeindex'].
        Returns Series indexed by ticker with percentile rank in [0, 100] (NaN for insufficient history).
        """
        universe = set(u.upper() for u in universe)
        df = panel[(panel["tradingdate"] <= as_of_date) &
                   (panel["stockcode"].str.upper().isin(universe))].copy()
        df["stockcode"] = df["stockcode"].str.upper()
        df = df.sort_values(["stockcode", "tradingdate"])
        raws = {}
        for t, g in df.groupby("stockcode"):
            raws[t] = _raw_rs(g["closeindex"].reset_index(drop=True), config)
        raw = pd.Series(raws, name="rs_raw")
        # Percentile-rank within the universe; NaNs stay NaN.
        ranked = raw.rank(pct=True, na_option="keep") * 100.0
        return ranked.rename("rs_rating")
    ```
