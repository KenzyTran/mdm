    ```python
    def _worker(args: Tuple[Dict[str, Any], Any]) -> Dict[str, Any]:
        config, _precomputed = args
        try:
            momentum_cfg = MomentumScorerConfig(
                rs_threshold=float(config["rs_threshold"]),
                n_within_high=float(config["n_within_high"]),
            )
            portfolio_cfg = PortfolioConfig(
                max_slots=int(config["slots"]),
                hard_stop_pct=float(config["hard_stop"]),
            )
            result = run_v8_backtest(
                momentum_cfg,
                portfolio_cfg,
                config["entry_option"],
                PERIOD,
                precomputed=None,   # each worker fetches from cache independently
                formula=config["rs_formula"],
            )
            metrics = result["metrics"]
            sanity = "OK" if metrics.get("CAGR", 0) < SANITY_CAGR_THRESHOLD else "CAGR_TOO_HIGH"
            row = {**config, **{k: metrics.get(k, float("nan")) for k in METRIC_COLS}, "sanity": sanity}
        except Exception as exc:
            print(f"[WARN] config {config} failed: {exc}", flush=True)
            row = {**config, **_nan_metrics(), "sanity": "ERROR"}
        return row
    ```

    **Main function:**
    ```python
    def main() -> None:
        grid = build_grid()
        print(f"Sweep: {len(grid)} configs, period {PERIOD[0]}..{PERIOD[1]}", flush=True)

        # Prime cache once before spawning workers (Pitfall 4 — avoids N-way DB race)
        print("Priming precompute cache...", flush=True)
        precompute_static(PERIOD)
        print("Cache primed.", flush=True)

        workers = max(1, cpu_count() - 1)
        args = [(cfg, None) for cfg in grid]

        try:
            from tqdm import tqdm
            with Pool(workers) as pool:
                rows = list(tqdm(pool.imap(_worker, args), total=len(args), desc="sweep_v8"))
        except ImportError:
            with Pool(workers) as pool:
                rows = pool.map(_worker, args)

        df = pd.DataFrame(rows, columns=GRID_COLS + METRIC_COLS + ["sanity"])
        OUTPUT_CSV.parent.mkdir(parents=True, exist_ok=True)
        df.to_csv(OUTPUT_CSV, index=False)
        print(f"Sweep saved to {OUTPUT_CSV} ({len(df)} rows)", flush=True)

        # Select winner: highest Sharpe_rf3 where CAGR>=5% and MaxDD>=-30%
        sane = df[(df["CAGR"] >= 0.05) & (df["MaxDD"] >= -0.30) & df["Sharpe_rf3"].notna()]
        if sane.empty:
            print("[WARN] No config passed sanity gates — using global best Sharpe", flush=True)
            sane = df[df["Sharpe_rf3"].notna()]
        best = sane.sort_values("Sharpe_rf3", ascending=False).iloc[0]
        locked = best[GRID_COLS + METRIC_COLS].to_dict()
        LOCKED_PARAMS.parent.mkdir(parents=True, exist_ok=True)
        with open(LOCKED_PARAMS, "w") as f:
            json.dump(locked, f, indent=2, default=float)
        print(f"Locked params written to {LOCKED_PARAMS}", flush=True)
        print(f"Winner: formula={locked['rs_formula']}, Sharpe_rf3={locked['Sharpe_rf3']:.3f}, CAGR={locked['CAGR']:.1%}, MaxDD={locked['MaxDD']:.1%}")

    if __name__ == "__main__":
        main()
    ```
